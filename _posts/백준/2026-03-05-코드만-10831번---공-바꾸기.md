```python
# 바구니에는 1~N번 번호
# 바구니 1, 바구니 2, 바구니 3..
# 바구니 내부의 초기 공 번호도 동일함.

# 첫째줄에 바구니 수, 공 바꿀 횟수가 나온다.
basket = list()

n, m = map(int, input().split()) # 초기 설정
for i in range(n):
    basket.append(i+1) # 바구니 안에 공 넣어두기

for j in range(m): # m번 동안..
    a, b = map(int, input().split()) # a-1, b-1번 바구니 배열 안에 있는 공을 바꾼다.
    tmp = basket[a-1]
    basket[a-1], basket[b-1] = basket[b-1], basket[a-1]

for b in basket:
    print(b, end=" ")

```
