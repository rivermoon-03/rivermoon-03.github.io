# 코드 작성
```python
# 어떤 자연수 M의 분해합이 N인 경우, M을 N의 생성자라 한다.
# (M)245의 분해합이 (N)256인 경우, (M)245를 (N)256의 생성자라 한다는 것이다.
# 245 + 2 + 4 + 5 = 256
# 256 - 245 - 2 - 4 - 5 = 0

# 256과 같은 숫자를 받았을 때, 256의 가장 작은 생성자를 구해내는 프로그램을 만들어야 한다는 것이다.

n = int(input()) # 분해합 (예: 256)
res = 0 # 실험쥐
result = 0 # 가장 작은 생성자

for res in range(1, n + 1): # 1부터 분해합 숫자까지 테스트해본다.
    m = res # m = 테스트할 숫자. (245까지 왔다고 가정하면, 일단 m = 245인것이다.)
    for num in str(res): # 자릿수를 더해본다.
        m += int(num) 
        # for문을 거치면서, 245 + 2 + 4 + 5 = 256이 m의 값이 된 것이다.

    if m == n: # 그러면 m과 n은 같을 것이다.
        result = res # 그러면 테스트한 숫자 (245)가 생성자가 맞다고 할 수 있는 것이다.
        break

print(result)

```

직접 하나하나 다 돌려보면서 테스트를 하는 브루트 포스 문제였다.
